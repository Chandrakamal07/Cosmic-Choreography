    <script>
        // Updated Planetary Data (Added Uranus & Neptune, Distinct Colors)
        const planets = {
            "Mercury": { R: 0.387, T: 0.241, color: "#a8a29e" },
            "Venus":   { R: 0.723, T: 0.615, color: "#f59e0b" },
            "Earth":   { R: 1.000, T: 1.000, color: "#3b82f6" }, // Explicit Blue
            "Mars":    { R: 1.524, T: 1.881, color: "#ef4444" },
            "Jupiter": { R: 5.204, T: 11.862, color: "#f97316" },
            "Saturn":  { R: 9.582, T: 29.457, color: "#fef08a" },
            "Uranus":  { R: 19.220, T: 84.01, color: "#2dd4bf" }, // Cyan/Teal
            "Neptune": { R: 30.050, T: 164.8, color: "#4f46e5" }  // Deep Indigo
        };

        // DOM Elements
        const container = document.getElementById('canvasContainer');
        const simCanvas = document.getElementById('simCanvas');
        const starsCanvas = document.getElementById('starsCanvas');
        const graphCanvas = document.getElementById('distanceGraph');
        
        const ctx = simCanvas.getContext('2d');
        const starsCtx = starsCanvas.getContext('2d');
        const graphCtx = graphCanvas.getContext('2d');

        const p1Select = document.getElementById('planet1');
        const p2Select = document.getElementById('planet2');
        const refFrameSelect = document.getElementById('refFrame');
        const speedSlider = document.getElementById('speed');
        const maxLinesSlider = document.getElementById('maxLines');
        const linesDisplay = document.getElementById('linesDisplay');
        const playPauseBtn = document.getElementById('playPauseBtn');
        const clearBtn = document.getElementById('clearBtn');
        const exportBtn = document.getElementById('exportBtn');
        const autoRemoveCheckbox = document.getElementById('autoRemove');
        const targetYearsInput = document.getElementById('targetYears');
        
        const simTimeDisplay = document.getElementById('simTimeDisplay');
        const legP1Color = document.getElementById('legP1Color');
        const legP1Name = document.getElementById('legP1Name');
        const legP2Color = document.getElementById('legP2Color');
        const legP2Name = document.getElementById('legP2Name');

        // State variables
        let time = 0;
        let isPlaying = false; 
        let drawnLines = [];
        let distanceHistory = []; // For the graph
        let animationId;
        let starsGenerated = false;

        // Initialize Dropdowns
        Object.keys(planets).forEach(p => {
            p1Select.add(new Option(p, p, false, p === "Earth"));
            p2Select.add(new Option(p, p, false, p === "Venus"));
        });

        // Resize Canvas to fit screen dynamically
        function resizeCanvases() {
            const rect = container.getBoundingClientRect();
            simCanvas.width = rect.width;
            simCanvas.height = rect.height;
            starsCanvas.width = rect.width;
            starsCanvas.height = rect.height;
            
            const graphRect = graphCanvas.parentElement.getBoundingClientRect();
            graphCanvas.width = graphRect.width;
            graphCanvas.height = graphRect.height;

            generateStars();
        }
        window.addEventListener('resize', resizeCanvases);

        function generateStars() {
            starsCtx.clearRect(0, 0, starsCanvas.width, starsCanvas.height);
            starsCtx.fillStyle = '#ffffff';
            for (let i = 0; i < 300; i++) {
                let x = Math.random() * starsCanvas.width;
                let y = Math.random() * starsCanvas.height;
                let size = Math.random() * 1.5;
                let opacity = Math.random() * 0.8 + 0.1;
                starsCtx.globalAlpha = opacity;
                starsCtx.beginPath();
                starsCtx.arc(x, y, size, 0, Math.PI * 2);
                starsCtx.fill();
            }
            starsCtx.globalAlpha = 1.0;
        }

        function resetSimulation() {
            time = 0;
            drawnLines = [];
            distanceHistory = [];
            isPlaying = false;
            playPauseBtn.innerText = "Play";
            playPauseBtn.style.backgroundColor = "var(--accent)";
            
            const p1 = p1Select.value;
            const p2 = p2Select.value;
            legP1Name.innerText = p1;
            legP1Color.style.background = planets[p1].color;
            legP1Color.style.color = planets[p1].color;
            legP2Name.innerText = p2;
            legP2Color.style.background = planets[p2].color;
            legP2Color.style.color = planets[p2].color;
        }

        // Draw Live Graph
        function drawGraph() {
            graphCtx.clearRect(0, 0, graphCanvas.width, graphCanvas.height);
            if (distanceHistory.length < 2) return;

            const maxDist = Math.max(...distanceHistory) * 1.1;
            const stepX = graphCanvas.width / 200; // Track last 200 points visually

            graphCtx.beginPath();
            graphCtx.strokeStyle = "#10b981"; // Emerald green graph
            graphCtx.lineWidth = 2;

            for (let i = 0; i < distanceHistory.length; i++) {
                let x = graphCanvas.width - ((distanceHistory.length - 1 - i) * stepX);
                let y = graphCanvas.height - (distanceHistory[i] / maxDist) * graphCanvas.height;
                
                if (i === 0) graphCtx.moveTo(x, y);
                else graphCtx.lineTo(x, y);
            }
            graphCtx.stroke();
            
            // Fill under graph
            graphCtx.lineTo(graphCanvas.width, graphCanvas.height);
            graphCtx.lineTo(graphCanvas.width - (distanceHistory.length * stepX), graphCanvas.height);
            graphCtx.fillStyle = "rgba(16, 185, 129, 0.1)";
            graphCtx.fill();
        }

        function draw() {
            if (isPlaying) {
                time += parseFloat(speedSlider.value);
                
                const target = parseFloat(targetYearsInput.value);
                if (!isNaN(target) && target > 0 && time >= target) {
                    time = target; 
                    isPlaying = false;
                    playPauseBtn.innerText = "Play";
                    playPauseBtn.style.backgroundColor = "var(--accent)";
                }
            }

            simTimeDisplay.innerText = `Simulated Time: ${time.toFixed(2)} Years`;

            // Clear Sim Canvas (Transparent so stars show through)
            ctx.clearRect(0, 0, simCanvas.width, simCanvas.height);

            const p1Data = planets[p1Select.value];
            const p2Data = planets[p2Select.value];
            const frame = refFrameSelect.value;
            
            const w1 = (2 * Math.PI) / p1Data.T;
            const w2 = (2 * Math.PI) / p2Data.T;

            const maxR = Math.max(p1Data.R, p2Data.R);
            const neededRadius = frame === 'heliocentric' ? maxR : (p1Data.R + p2Data.R);
            
            // Calculate scale to fit within the shortest dimension of the canvas
            const minDim = Math.min(simCanvas.width, simCanvas.height);
            const scale = (minDim / 2) / (neededRadius * 1.15); 
            
            const cx = simCanvas.width / 2;
            const cy = simCanvas.height / 2;

            let sunPos, p1Pos, p2Pos;

            if (frame === 'heliocentric') {
                sunPos = { x: 0, y: 0 };
                p1Pos = { x: p1Data.R * Math.cos(w1 * time), y: p1Data.R * Math.sin(w1 * time) };
                p2Pos = { x: p2Data.R * Math.cos(w2 * time), y: p2Data.R * Math.sin(w2 * time) };
            } else {
                p1Pos = { x: 0, y: 0 };
                sunPos = { x: -p1Data.R * Math.cos(w1 * time), y: -p1Data.R * Math.sin(w1 * time) };
                p2Pos = { x: sunPos.x + p2Data.R * Math.cos(w2 * time), y: sunPos.y + p2Data.R * Math.sin(w2 * time) };
            }

            if (isPlaying) {
                drawnLines.push({
                    x1: p1Pos.x, y1: p1Pos.y,
                    x2: p2Pos.x, y2: p2Pos.y
                });
                
                // Track distance for graph
                const dist = Math.hypot(p1Pos.x - p2Pos.x, p1Pos.y - p2Pos.y);
                distanceHistory.push(dist);
                if (distanceHistory.length > 200) distanceHistory.shift();
            }

            if (autoRemoveCheckbox.checked) {
                const maxHistory = parseInt(maxLinesSlider.value);
                while (drawnLines.length > maxHistory) {
                    drawnLines.shift();
                }
            }

            drawGraph();

            // Draw Base Orbits
            if (frame === 'heliocentric') {
                ctx.beginPath(); ctx.arc(cx, cy, p1Data.R * scale, 0, 2 * Math.PI);
                ctx.strokeStyle = "rgba(255,255,255,0.08)"; ctx.stroke();
                ctx.beginPath(); ctx.arc(cx, cy, p2Data.R * scale, 0, 2 * Math.PI);
                ctx.stroke();
            }

            // Draw Resonance Lines
            ctx.lineWidth = 0.5;
            ctx.strokeStyle = "rgba(255, 255, 255, 0.25)";
            ctx.beginPath();
            drawnLines.forEach(line => {
                ctx.moveTo(cx + line.x1 * scale, cy + line.y1 * scale);
                ctx.lineTo(cx + line.x2 * scale, cy + line.y2 * scale);
            });
            ctx.stroke();

            // Draw Celestial Bodies
            function drawBody(pos, color, radius) {
                ctx.beginPath();
                ctx.arc(cx + pos.x * scale, cy + pos.y * scale, radius, 0, 2 * Math.PI);
                ctx.fillStyle = color;
                ctx.shadowColor = color;
                ctx.shadowBlur = 20;
                ctx.fill();
                ctx.shadowBlur = 0; 
            }
            drawBody(sunPos, "#fbbf24", 8);
            drawBody(p1Pos, p1Data.color, 5);
            drawBody(p2Pos, p2Data.color, 5);

            animationId = requestAnimationFrame(draw);
        }

        // Event Listeners
        p1Select.addEventListener('change', resetSimulation);
        p2Select.addEventListener('change', resetSimulation);
        refFrameSelect.addEventListener('change', resetSimulation);
        
        maxLinesSlider.addEventListener('input', (e) => {
            linesDisplay.innerText = e.target.value;
            if (autoRemoveCheckbox.checked) {
                while (drawnLines.length > parseInt(e.target.value)) {
                    drawnLines.shift();
                }
            }
        });

        clearBtn.addEventListener('click', () => { drawnLines = []; distanceHistory = []; });

        playPauseBtn.addEventListener('click', () => {
            if (!isPlaying && targetYearsInput.value && time >= parseFloat(targetYearsInput.value)) {
                targetYearsInput.value = ''; 
            }
            isPlaying = !isPlaying;
            playPauseBtn.innerText = isPlaying ? "Pause" : "Play";
            if(isPlaying) {
                playPauseBtn.style.backgroundColor = "#ef4444"; 
            } else {
                playPauseBtn.style.backgroundColor = "var(--accent)"; 
            }
        });

        // Export Functionality (Combines Stars + Sim Canvas)
        exportBtn.addEventListener('click', () => {
            const tempCanvas = document.createElement('canvas');
            tempCanvas.width = simCanvas.width;
            tempCanvas.height = simCanvas.height;
            const tCtx = tempCanvas.getContext('2d');
            
            // Draw gradient background
            const grad = tCtx.createRadialGradient(tempCanvas.width/2, tempCanvas.height/2, 0, tempCanvas.width/2, tempCanvas.height/2, tempCanvas.width);
            grad.addColorStop(0, '#0f172a');
            grad.addColorStop(1, '#000000');
            tCtx.fillStyle = grad;
            tCtx.fillRect(0, 0, tempCanvas.width, tempCanvas.height);
            
            // Draw Stars and Sim
            tCtx.drawImage(starsCanvas, 0, 0);
            tCtx.drawImage(simCanvas, 0, 0);
            
            // Draw Legend manually for export
            const legX = tempCanvas.width - 220;
            const legY = 30;
            tCtx.fillStyle = "rgba(15, 23, 42, 0.85)";
            tCtx.beginPath(); tCtx.roundRect(legX, legY, 190, 140, 12); tCtx.fill();
            tCtx.strokeStyle = "rgba(255,255,255,0.2)"; tCtx.stroke();
            
            tCtx.font = "bold 16px system-ui"; tCtx.fillStyle = "#fff";
            tCtx.fillText(`Time: ${time.toFixed(2)} Years`, legX + 20, legY + 30);
            
            tCtx.font = "16px system-ui";
            tCtx.fillStyle = "#fbbf24"; tCtx.beginPath(); tCtx.arc(legX + 25, legY + 65, 6, 0, Math.PI*2); tCtx.fill();
            tCtx.fillStyle = "#fff"; tCtx.fillText("Sun", legX + 45, legY + 70);
            
            tCtx.fillStyle = planets[p1Select.value].color; tCtx.beginPath(); tCtx.arc(legX + 25, legY + 95, 6, 0, Math.PI*2); tCtx.fill();
            tCtx.fillStyle = "#fff"; tCtx.fillText(p1Select.value, legX + 45, legY + 100);
            
            tCtx.fillStyle = planets[p2Select.value].color; tCtx.beginPath(); tCtx.arc(legX + 25, legY + 125, 6, 0, Math.PI*2); tCtx.fill();
            tCtx.fillStyle = "#fff"; tCtx.fillText(p2Select.value, legX + 45, legY + 130);

            const link = document.createElement('a');
            link.download = `Orbital_Resonance_${p1Select.value}_${p2Select.value}.png`;
            link.href = tempCanvas.toDataURL('image/png');
            link.click();
        });

        // Start
        resizeCanvases();
        resetSimulation();
        draw(); 

    </script>
